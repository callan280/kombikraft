#!/usr/bin/env python3
from pathlib import Path
import json,hashlib,struct,shutil
HERE=Path(__file__).resolve().parent;LAB=HERE.parents[1];PACKAGE=LAB/'f97-reference/render-package/f97-gauges-package.json'
EXPECTED='c81132e977a9de407997893fd0a177f1c969ddd8732efbac6058b306d4a05864'
sha=lambda b:hashlib.sha256(b).hexdigest()
assert sha(PACKAGE.read_bytes())==EXPECTED,'Source package changed'
d=json.loads(PACKAGE.read_text());OUT=HERE/'stage/f97';OUT.mkdir(parents=True,exist_ok=True)
rows=[];lines=['/* Original source geometry; generated by prepare_assets.py. */']
for k,m in d['meshes'].items():
 assert m['vertexCount']==len(m['originalPosition'])==len(m['originalUv0']) and m['vertexCount']%3==0
 p=[float(v) for xyz in m['originalPosition'] for v in xyz];uv=[float(v) for values in m['originalUv0'] for v in values[:2]]
 assert len(p)==m['vertexCount']*3 and m['vertexCount']<65536
 for attr,data in [('positions',p),('uv',uv)]:
  b=struct.pack('<'+'f'*len(data),*data);file=OUT/(k+'-'+attr+'.f32');file.write_bytes(b)
  lines.append('static float '+k+'_'+attr+'[]={'+','.join(format(v,'.9g')+('f' if '.' in format(v,'.9g') or 'e' in format(v,'.9g') else '.0f') for v in data)+'};')
  rows.append({'kind':'original-'+attr,'mesh':k,'file':str(file.relative_to(HERE)), 'sha256':sha(b),'vertices':m['vertexCount'],'sourceMeshJson':m['sourceMeshJson'],'sourceDecodedSha256':m['sourceDecodedSha256']})
textures=[]
for k,t in d['textures'].items():
 b=(PACKAGE.parent/t['file']).read_bytes();assert sha(b)==t['sha256']
 if t['storedBytesPerPixel']==2:
  rgba=bytearray()
  for i in range(0,len(b),2):rgba.extend((b[i],b[i],b[i],b[i+1]))
  b=bytes(rgba);conversion='L,A to L,L,L,A; explicit custom preview binding, not vendor enum proof'
 else:conversion='Original four stored bytes and row order, no conversion'
 f=OUT/(k+'.rgba');f.write_bytes(b);assert len(b)==t['width']*t['height']*4
 textures.append({'key':k,'file':f.name,'width':t['width'],'height':t['height'],'sha256':sha(b),'sourceSha256':t['sha256'],'conversion':conversion})
lines.append('typedef struct{const char *key,*file;unsigned width,height;} F97Texture;')
lines.append('static const F97Texture f97_textures[]={'+','.join('{"'+t['key']+'","'+t['file']+'",'+str(t['width'])+','+str(t['height'])+'}' for t in textures)+'};')
(HERE/'f97_arrays.inc').write_text('\n'.join(lines)+'\n')
(HERE/'asset-manifest.json').write_text(json.dumps({'packageSha256':EXPECTED,'sourcePackage':str(PACKAGE),'sourcePackageUnmodified':True,'meshArrays':rows,'textures':textures,'nativeGeometryMaterial':'Custom unlit GLSL port, not original compiled F97 SHDR material','originalControllerCalibrationVerified':False,'physicalF40MaskVerified':False},indent=2)+'\n')
shutil.copyfile(PACKAGE,OUT/'source-package.json')
print(len(rows),'arrays',len(textures),'textures')
